from boleto_extractor.core.exceptions import BoletoValidationError


class BarcodeToDigitableLine:
    """Converte código de barras de arrecadação em linha digitável."""

    def converter(self, codigo: str) -> str:
        self._validar_entrada(codigo)

        modulo = self._obter_modulo(codigo)

        if not self._validar_dv_geral(codigo, modulo):
            raise BoletoValidationError(
                "O dígito verificador geral do código de barras é inválido."
            )

        grupos = [
            codigo[i : i + 11]
            for i in range(0, 44, 11)
        ]

        grupos_formatados = [
            grupo + str(self._calcular_dv(grupo, modulo))
            for grupo in grupos
        ]

        return "".join(grupos_formatados)

    def _validar_entrada(self, codigo: str) -> None:
        if not isinstance(codigo, str):
            raise BoletoValidationError(
                "O código de barras precisa ser uma string."
            )

        if len(codigo) != 44:
            raise BoletoValidationError(
                "O código de barras precisa possuir 44 dígitos."
            )

        if not codigo.isdigit():
            raise BoletoValidationError(
                "O código de barras deve conter somente números."
            )

        if not codigo.startswith("8"):
            raise BoletoValidationError(
                "O código informado não é de arrecadação."
            )

        if codigo[2] not in "6789":
            raise BoletoValidationError(
                "Identificador de valor ou referência inválido."
            )

    def _obter_modulo(self, codigo: str) -> int:
        return 10 if codigo[2] in "67" else 11

    def _calcular_dv(self, numero: str, modulo: int) -> int:
        if modulo == 10:
            return self._calcular_modulo_10(numero)

        return self._calcular_modulo_11(numero)

    def _calcular_modulo_10(self, numero: str) -> int:
        soma = 0
        peso = 2

        for digito in reversed(numero):
            produto = int(digito) * peso
            soma += produto // 10 + produto % 10
            peso = 1 if peso == 2 else 2

        return (10 - soma % 10) % 10

    def _calcular_modulo_11(self, numero: str) -> int:
        soma = 0
        peso = 2

        for digito in reversed(numero):
            soma += int(digito) * peso
            peso = 2 if peso == 9 else peso + 1

        resto = soma % 11

        if resto in (0, 1):
            return 0

        if resto == 10:
            return 1

        return 11 - resto

    def _validar_dv_geral(
        self,
        codigo: str,
        modulo: int,
    ) -> bool:
        base = codigo[:3] + codigo[4:]
        dv_informado = int(codigo[3])

        return self._calcular_dv(base, modulo) == dv_informado

import re
from dataclasses import dataclass

from boleto_extractor.application.services.documento_validator import (
    cnpj_valido,
    cpf_valido,
    limpar,
)

_RE_CNPJ = re.compile(
    r"(?<![A-Z0-9])[A-Z0-9]{2}\.?[A-Z0-9]{3}\.?[A-Z0-9]{3}/?[A-Z0-9]{4}-?\d{2}(?![A-Z0-9])"
)
_RE_CPF = re.compile(r"(?<![A-Z0-9])\d{3}\.?\d{3}\.?\d{3}-?\d{2}(?![A-Z0-9])")

_ROTULOS: tuple[tuple[str, re.Pattern[str]], ...] = (
    ("beneficiario", re.compile(r"BENEFICI[AÁ]RIO|CEDENTE")),
    ("avalista", re.compile(r"SACADOR|AVALISTA")),
    ("pagador", re.compile(r"PAGADOR|\bSACADO\b")),
)
_JANELA = 400


@dataclass(frozen=True, slots=True)
class DocumentoEncontrado:
    documento: str
    tipo: str
    papel: str
    posicao: int
    confirmado_pelo_codigo: bool = False


class BeneficiarioExtractor:
    """Localiza o CPF/CNPJ do beneficiário no texto do boleto."""

    def extrair(self, texto: str, raiz_cnpj: str | None = None) -> DocumentoEncontrado | None:
        try:
            docs = self._todos(texto)
        except Exception:
            return None

        if raiz_cnpj:
            for d in docs:
                if d.tipo == "CNPJ" and d.documento.startswith(raiz_cnpj):
                    return DocumentoEncontrado(
                        d.documento, d.tipo, d.papel, d.posicao, confirmado_pelo_codigo=True
                    )

        criterios = (
            lambda d: d.papel == "beneficiario",
            lambda d: d.papel == "desconhecido" and d.tipo == "CNPJ",
            lambda d: d.papel == "desconhecido",
        )
        for criterio in criterios:
            for d in docs:
                if criterio(d):
                    return d
        return None

    def _todos(self, texto: str) -> list[DocumentoEncontrado]:
        t = (texto or "").upper()
        rotulos = sorted((m.start(), p) for p, rx in _ROTULOS for m in rx.finditer(t))

        def papel_de(pos: int) -> str:
            antes = [r for p, r in rotulos if p < pos and pos - p <= _JANELA]
            return antes[-1] if antes else "desconhecido"

        achados: list[DocumentoEncontrado] = []
        vistos: set[str] = set()
        buscas = ((_RE_CNPJ, "CNPJ", cnpj_valido), (_RE_CPF, "CPF", cpf_valido))
        for rx, tipo, valida in buscas:
            for m in rx.finditer(t):
                doc = limpar(m.group())
                if doc in vistos or not valida(doc):
                    continue
                vistos.add(doc)
                achados.append(DocumentoEncontrado(doc, tipo, papel_de(m.start()), m.start()))
        return sorted(achados, key=lambda d: d.posicao)

from collections.abc import Callable
from datetime import date, timedelta
from decimal import Decimal

from boleto_extractor.domain.entities.boleto_details import BoletoDetails

BASE_FATOR_ANTIGA = date(1997, 10, 7)   # fator 1000 = 03/07/2000 ... 9999 = 21/02/2025
BASE_FATOR_NOVA = date(2025, 2, 22)     # fator 1000 = 22/02/2025 (reinício Febraban)

BANCOS: dict[str, str] = {
    "001": "Banco do Brasil", "004": "Banco do Nordeste", "021": "Banestes",
    "033": "Santander", "037": "Banpará", "041": "Banrisul", "047": "Banese",
    "070": "BRB", "077": "Inter", "104": "Caixa", "136": "Unicred",
    "208": "BTG Pactual", "212": "Original", "237": "Bradesco",
    "260": "Nubank", "290": "PagBank", "318": "BMG", "323": "Mercado Pago",
    "336": "C6 Bank", "341": "Itaú", "380": "PicPay", "389": "Mercantil",
    "403": "Cora", "422": "Safra", "633": "Rendimento", "655": "Votorantim",
    "707": "Daycoval", "745": "Citibank", "748": "Sicredi", "756": "Sicoob",
}

SEGMENTOS: dict[str, str] = {
    "1": "Prefeitura", "2": "Saneamento", "3": "Energia elétrica e gás",
    "4": "Telecomunicações", "5": "Órgão governamental",
    "6": "Empresa (CNPJ)", "7": "Multa de trânsito", "9": "Uso exclusivo do banco",
}


class BoletoParser:
    """Extrai valor, vencimento e emissor de um código de barras de 44 dígitos."""

    def __init__(self, referencia: Callable[[], date] | None = None) -> None:
        self._referencia = referencia or date.today

    # ---------------- Cobrança ----------------
    def cobranca(self, codigo: str) -> BoletoDetails:
        banco = codigo[0:3]
        observacoes: list[str] = []

        vencimento = self._data_por_fator(codigo[5:9])
        if vencimento is None:
            observacoes.append("Boleto sem vencimento definido (fator 0000).")

        valor = Decimal(int(codigo[9:19])) / 100
        if valor == 0:
            observacoes.append("Valor não informado no código (valor em aberto).")

        return BoletoDetails(
            valor=valor or None,
            vencimento=vencimento,
            banco_codigo=banco,
            banco_nome=BANCOS.get(banco, f"Banco {banco}"),
            campo_livre=codigo[19:44],
            observacoes=tuple(observacoes),
        )

    def _data_por_fator(self, fator_txt: str) -> date | None:
        fator = int(fator_txt)
        if fator < 1000:
            return None

        candidatas = [BASE_FATOR_NOVA + timedelta(days=fator - 1000)]
        candidatas.append(BASE_FATOR_ANTIGA + timedelta(days=fator))

        hoje = self._referencia()
        return min(candidatas, key=lambda d: abs((d - hoje).days))

    # ---------------- Arrecadação ----------------
    def arrecadacao(self, codigo: str) -> BoletoDetails:
        segmento = codigo[1]
        identificador = codigo[2]
        observacoes: list[str] = []

        valor: Decimal | None = None
        if identificador in "68":
            valor = (Decimal(int(codigo[4:15])) / 100) or None
        else:
            observacoes.append("Código usa quantidade de referência, não valor em reais.")

        tamanho_empresa = 8 if segmento == "6" else 4
        empresa = codigo[15:15 + tamanho_empresa]
        livre = codigo[15 + tamanho_empresa:44]

        vencimento = self._data_no_campo_livre(livre)
        if vencimento:
            observacoes.append("Vencimento estimado a partir do campo livre.")

        return BoletoDetails(
            valor=valor,
            vencimento=vencimento,
            vencimento_estimado=vencimento is not None,
            segmento=SEGMENTOS.get(segmento, f"Segmento {segmento}"),
            empresa_id=empresa,
            campo_livre=livre,
            observacoes=tuple(observacoes),
        )

    def _data_no_campo_livre(self, livre: str) -> date | None:
        hoje = self._referencia()
        limite = timedelta(days=730)

        for i in range(len(livre) - 7):
            trecho = livre[i:i + 8]
            if not trecho.startswith("20"):
                continue
            try:
                data = date(int(trecho[0:4]), int(trecho[4:6]), int(trecho[6:8]))
            except ValueError:
                continue
            if abs(data - hoje) <= limite:
                return data

        return None

from boleto_extractor.core.exceptions import BoletoValidationError


class BarcodeToDigitableLine:
    """Converte código de barras bancário em linha digitável e vice-versa."""

    def converter(self, codigo: str) -> str:
        self._validar_entrada(codigo)

        if not self._validar_dv_geral(codigo):
            raise BoletoValidationError("O dígito verificador geral é inválido.")

        campo_1 = codigo[:4] + codigo[19:24]
        campo_2 = codigo[24:34]
        campo_3 = codigo[34:44]

        return (
            campo_1 + str(self._calcular_modulo_10(campo_1))
            + campo_2 + str(self._calcular_modulo_10(campo_2))
            + campo_3 + str(self._calcular_modulo_10(campo_3))
            + codigo[4]
            + codigo[5:19]
        )

    def converter_para_codigo_barras(self, linha: str) -> str:
        """Converte uma linha digitável de 47 dígitos em código de barras de 44."""
        self._validar_linha_digitavel(linha)

        campos = (
            (linha[0:9], int(linha[9]), "primeiro"),
            (linha[10:20], int(linha[20]), "segundo"),
            (linha[21:31], int(linha[31]), "terceiro"),
        )

        for campo, dv, nome in campos:
            if self._calcular_modulo_10(campo) != dv:
                raise BoletoValidationError(
                    f"O dígito verificador do {nome} campo é inválido."
                )

        codigo = (
            linha[0:4]
            + linha[32]
            + linha[33:47]
            + linha[4:9]
            + linha[10:20]
            + linha[21:31]
        )

        self._validar_entrada(codigo)

        if not self._validar_dv_geral(codigo):
            raise BoletoValidationError("O dígito verificador geral é inválido.")

        return codigo

    def _validar_entrada(self, codigo: str) -> None:
        if not isinstance(codigo, str):
            raise BoletoValidationError("O código precisa ser uma string.")

        if len(codigo) != 44:
            raise BoletoValidationError("O código precisa possuir 44 dígitos.")

        if not codigo.isdigit():
            raise BoletoValidationError("O código deve conter somente números.")

        if codigo[3] != "9":
            raise BoletoValidationError(
                "Código de moeda incompatível com boleto em Real."
            )

    def _validar_linha_digitavel(self, linha: str) -> None:
        if not isinstance(linha, str):
            raise BoletoValidationError("A linha digitável precisa ser uma string.")

        if len(linha) != 47:
            raise BoletoValidationError("A linha digitável precisa possuir 47 dígitos.")

        if not linha.isdigit():
            raise BoletoValidationError(
                "A linha digitável deve conter somente números."
            )

    def _calcular_modulo_10(self, numero: str) -> int:
        soma = 0
        peso = 2

        for digito in reversed(numero):
            produto = int(digito) * peso
            soma += produto // 10 + produto % 10
            peso = 1 if peso == 2 else 2

        return (10 - soma % 10) % 10

    def _calcular_modulo_11(self, numero: str) -> int:
        soma = 0
        peso = 2

        for digito in reversed(numero):
            soma += int(digito) * peso
            peso = 2 if peso == 9 else peso + 1

        resultado = 11 - soma % 11

        return 1 if resultado in (0, 10, 11) else resultado

    def _validar_dv_geral(self, codigo: str) -> bool:
        base = codigo[:4] + codigo[5:]
        return self._calcular_modulo_11(base) == int(codigo[4])

class CodeNormalizer:
    """Converte qualquer entrada (44, 47 ou 48 dígitos) para o código de barras de 44."""

    def __init__(
        self,
        cobranca: CobrancaConverter | None = None,
        arrecadacao: ArrecadacaoConverter | None = None,
    ) -> None:
        self._cobranca = cobranca or CobrancaConverter()
        self._arrecadacao = arrecadacao or ArrecadacaoConverter()

    def para_codigo_barras(self, texto: str) -> str:
        digitos = "".join(c for c in texto if c.isdigit())
        tamanho = len(digitos)

        if tamanho == 44:
            return digitos

        if tamanho == 47:
            if digitos.startswith("8"):
                raise BoletoValidationError("Linha de 47 dígitos não pode começar com 8.")
            codigo = (
                digitos[0:4] + digitos[32] + digitos[33:47]
                + digitos[4:9] + digitos[10:20] + digitos[21:31]
            )
            self._conferir(self._cobranca.converter(codigo), digitos)
            return codigo

        if tamanho == 48:
            if not digitos.startswith("8"):
                raise BoletoValidationError("Linha de arrecadação deve começar com 8.")
            codigo = digitos[0:11] + digitos[12:23] + digitos[24:35] + digitos[36:47]
            self._conferir(self._arrecadacao.converter(codigo), digitos)
            return codigo

        if tamanho == 0:
            raise BoletoValidationError("Nenhum número informado.")

        raise BoletoValidationError(
            f"Código com {tamanho} dígitos. Esperado 44, 47 ou 48."
        )

    @staticmethod
    def _conferir(recalculada: str, informada: str) -> None:
        if recalculada != informada:
            raise BoletoValidationError(
                "Dígito verificador de campo não confere. Verifique a digitação."
            )

import re

_NAO_ALNUM = re.compile(r"[^A-Z0-9]")


def limpar(valor: str) -> str:
    return _NAO_ALNUM.sub("", (valor or "").upper())


def cpf_valido(cpf: str) -> bool:
    d = re.sub(r"\D", "", cpf or "")
    if len(d) != 11 or d == d[0] * 11:
        return False
    for n in (9, 10):
        soma = sum(int(d[i]) * (n + 1 - i) for i in range(n))
        if (soma * 10 % 11) % 10 != int(d[n]):
            return False
    return True


def _dv_cnpj(base: str) -> str:
    """DV do CNPJ numérico ou alfanumérico (valor = ASCII - 48; pesos 2..9)."""
    soma = sum((ord(c) - 48) * ((i % 8) + 2) for i, c in enumerate(reversed(base)))
    resto = soma % 11
    return "0" if resto < 2 else str(11 - resto)


def cnpj_valido(cnpj: str) -> bool:
    c = limpar(cnpj)
    if not re.fullmatch(r"[A-Z0-9]{12}\d{2}", c) or c == c[0] * 14:
        return False
    dv1 = _dv_cnpj(c[:12])
    return c[12:] == dv1 + _dv_cnpj(c[:12] + dv1)

from boleto_extractor.infrastructure.erp.erp_parcela_repository import ParcelaErp

_TOL = Decimal("0.01")


class StatusCruzamento(str, Enum):
    ENCONTRADO = "encontrado"
    JA_PAGO = "ja_pago"
    DATA_DIVERGENTE = "data_divergente"
    VALOR_DIVERGENTE = "valor_divergente"
    DOC_INEXISTENTE = "doc_inexistente"
    NAO_ENCONTRADO = "nao_encontrado"
    DUPLICADO = "duplicado"


ROTULOS = {
    StatusCruzamento.ENCONTRADO: "Encontrado",
    StatusCruzamento.JA_PAGO: "Já pago",
    StatusCruzamento.DATA_DIVERGENTE: "Data divergente",
    StatusCruzamento.VALOR_DIVERGENTE: "Valor divergente",
    StatusCruzamento.DOC_INEXISTENTE: "Doc. inexistente",
    StatusCruzamento.NAO_ENCONTRADO: "Não encontrado",
    StatusCruzamento.DUPLICADO: "Duplicado",
}


@dataclass(frozen=True, slots=True)
class ItemCruzamento:
    chave: str          # identificador da linha (código de barras)
    documento: int      # pag_cod
    vencimento: date
    valor: Decimal


@dataclass(frozen=True, slots=True)
class ResultadoCruzamento:
    chave: str
    status: StatusCruzamento
    parcela: ParcelaErp | None
    detalhe: str


def _br(d: date | None) -> str:
    return d.strftime("%d/%m/%Y") if d else "--"


def _moeda(v: Decimal) -> str:
    return f"R$ {v:,.2f}".replace(",", "X").replace(".", ",").replace("X", ".")


def _valor_confere(p: ParcelaErp, valor: Decimal) -> bool:
    return abs(p.valor_documento - valor) <= _TOL or abs(p.saldo - valor) <= _TOL


def cruzar(itens: list[ItemCruzamento],
           parcelas_por_conta: dict[int, list[ParcelaErp]]) -> list[ResultadoCruzamento]:
    usadas: set[int] = set()
    saida: list[ResultadoCruzamento] = []

    for it in itens:
        parcelas = parcelas_por_conta.get(it.documento, [])
        if not parcelas:
            saida.append(ResultadoCruzamento(it.chave, StatusCruzamento.DOC_INEXISTENTE, None,
                                             f"Documento {it.documento} não existe no ERP"))
            continue

        exatas = [p for p in parcelas if p.vencimento == it.vencimento and _valor_confere(p, it.valor)]
        livres = [p for p in exatas if p.cod not in usadas]

        if livres:
            abertas = [p for p in livres if not p.pago]
            escolhida = (abertas or livres)[0]
            usadas.add(escolhida.cod)
            if escolhida.pago:
                saida.append(ResultadoCruzamento(it.chave, StatusCruzamento.JA_PAGO, escolhida,
                                                 "Parcela já baixada no ERP"))
            else:
                saida.append(ResultadoCruzamento(it.chave, StatusCruzamento.ENCONTRADO, escolhida,
                                                 f"Saldo {_moeda(escolhida.saldo)}"))
            continue

        if exatas:
            saida.append(ResultadoCruzamento(it.chave, StatusCruzamento.DUPLICADO, exatas[0],
                                             "Parcela já usada por outro boleto do lote"))
            continue

        mesmo_valor = [p for p in parcelas if _valor_confere(p, it.valor) and not p.pago]
        if mesmo_valor:
            datas = ", ".join(_br(p.vencimento) for p in mesmo_valor[:6])
            saida.append(ResultadoCruzamento(it.chave, StatusCruzamento.DATA_DIVERGENTE, None,
                                             f"Vencimentos com esse valor: {datas}"))
            continue

        mesma_data = [p for p in parcelas if p.vencimento == it.vencimento]
        if mesma_data:
            valores = ", ".join(_moeda(p.valor_documento) for p in mesma_data[:4])
            saida.append(ResultadoCruzamento(it.chave, StatusCruzamento.VALOR_DIVERGENTE, None,
                                             f"Valor no ERP: {valores}"))
            continue

        saida.append(ResultadoCruzamento(it.chave, StatusCruzamento.NAO_ENCONTRADO, None,
                                         f"{len(parcelas)} parcela(s) no documento, nenhuma confere"))
    return saida

from __future__ import annotations

import re
from datetime import date, datetime, timedelta

_BASE_ANTIGA = date(1997, 10, 7)    # fator 0000
_BASE_NOVA = date(2025, 2, 22)      # fator 1000 após o reinício
_RE_VENC = re.compile(r"venc\w*\D{0,30}?(\d{2})[/.-](\d{2})[/.-](\d{4})", re.I | re.S)


def _por_fator(codigo: str, hoje: date) -> date | None:
    fator = int(codigo[5:9])
    if fator == 0:
        return None
    antiga = _BASE_ANTIGA + timedelta(days=fator)
    nova = _BASE_NOVA + timedelta(days=fator - 1000)
    return min((antiga, nova), key=lambda d: abs((d - hoje).days))


def _por_texto(texto: str) -> date | None:
    for d, m, a in _RE_VENC.findall(texto or ""):
        try:
            return datetime(int(a), int(m), int(d)).date()
        except ValueError:
            continue
    return None


def _por_campo_livre(codigo: str, hoje: date) -> date | None:
    livre = codigo[19:]
    for i in range(len(livre) - 7):
        trecho = livre[i:i + 8]
        if not trecho.startswith("20"):
            continue
        try:
            d = datetime.strptime(trecho, "%Y%m%d").date()
        except ValueError:
            continue
        if abs((d - hoje).days) <= 366:
            return d
    return None


def extrair_vencimento(codigo_barras: str, texto_pdf: str = "",
                       hoje: date | None = None) -> tuple[date | None, str]:
    """Retorna (data, origem) com origem em: 'codigo', 'pdf' ou ''."""
    hoje = hoje or date.today()
    cod = re.sub(r"\D", "", codigo_barras or "")
    if len(cod) != 44:
        return _por_texto(texto_pdf), "pdf" if _por_texto(texto_pdf) else ""

    if not cod.startswith("8") and (d := _por_fator(cod, hoje)):
        return d, "codigo"
    if d := _por_texto(texto_pdf):
        return d, "pdf"
    if cod.startswith("8") and (d := _por_campo_livre(cod, hoje)):
        return d, "pdf"
    return None, ""

                arquivo=arquivo,
                status=ExtractionStatus.INVALID,
                mensagens=tuple(mensagens),

ORIGEM_MANUAL = Path("Leitor de código")


class ExtractBoletoUseCase:
    """Extrai boletos de documentos ou de códigos digitados/bipados."""

    def __init__(
        self,
        loader: DocumentLoader,
        reader: BarcodeReader,
        cobranca: CobrancaConverter | None = None,
        arrecadacao: ArrecadacaoConverter | None = None,
        parser: BoletoParser | None = None,
        text_extractor: TextExtractor | None = None,
        cnpj_lookup: CnpjLookup | None = None,
        beneficiario_extractor: BeneficiarioExtractor | None = None,
    ) -> None:
        self._loader = loader
        self._reader = reader
        self._cobranca = cobranca or CobrancaConverter()
        self._arrecadacao = arrecadacao or ArrecadacaoConverter()
        self._parser = parser or BoletoParser()
        self._normalizer = CodeNormalizer(self._cobranca, self._arrecadacao)
        self._text_extractor = text_extractor
        self._cnpj_lookup = cnpj_lookup
        self._benef_extractor = beneficiario_extractor or BeneficiarioExtractor()

    def executar(self, arquivo: Path) -> tuple[ExtractionResult, ...]:
        resultados: list[ExtractionResult] = []
        vistos: set[str] = set()

        try:
            for pagina, imagem in enumerate(self._loader.carregar(arquivo), start=1):
                for leitura in self._reader.ler(imagem):
                    codigo = "".join(c for c in leitura.valor if c.isdigit())

                    if len(codigo) != 44 or codigo in vistos:
                        continue
                    vistos.add(codigo)

                    resultados.append(self._processar(arquivo, pagina, codigo))

        except BoletoExtractorError as exc:
            resultados.append(
                ExtractionResult(
                    arquivo=arquivo,
                    status=ExtractionStatus.ERROR,
                    mensagens=(str(exc),),
                )
            )

        if not resultados:
            resultados.append(
                ExtractionResult(
                    arquivo=arquivo,
                    status=ExtractionStatus.REVIEW,
                    mensagens=("Nenhum código de barras encontrado.",),
                )
            )

        return tuple(self._enriquecer(arquivo, resultados))

    def executar_codigo(self, texto: str, origem: Path = ORIGEM_MANUAL) -> ExtractionResult:
        """Processa um código vindo do leitor USB ou digitado (44, 47 ou 48 dígitos)."""
        try:
            codigo = self._normalizer.para_codigo_barras(texto)
        except BoletoValidationError as exc:
            return ExtractionResult(
                arquivo=origem,
                status=ExtractionStatus.INVALID,
                mensagens=(str(exc),),
            )

        return self._processar(origem, None, codigo)

    # ---------------- Beneficiário ----------------
    def _enriquecer(
        self, arquivo: Path, resultados: list[ExtractionResult]
    ) -> list[ExtractionResult]:
        """Anexa o beneficiário. Qualquer falha mantém o resultado original."""
        if self._text_extractor is None:
            return resultados
        if not any(r.status == ExtractionStatus.SUCCESS for r in resultados):
            return resultados

        try:
            paginas = self._text_extractor.extrair_paginas(arquivo)
        except Exception:
            return resultados
        if not any(paginas):
            return resultados

        texto_total = "\n".join(paginas)
        saida: list[ExtractionResult] = []
        for r in resultados:
            texto = texto_total
            if r.pagina and 0 < r.pagina <= len(paginas) and paginas[r.pagina - 1]:
                texto = paginas[r.pagina - 1]
            saida.append(self._com_beneficiario(r, texto))
        return saida

    def _com_beneficiario(self, resultado: ExtractionResult, texto: str) -> ExtractionResult:
        if resultado.status != ExtractionStatus.SUCCESS or resultado.boleto is None:
            return resultado
        try:
            achado = self._benef_extractor.extrair(texto, self._raiz_cnpj(resultado.boleto))
            if achado is None:
                return resultado

            empresa = None
            if achado.tipo == "CNPJ" and self._cnpj_lookup is not None:
                empresa = self._cnpj_lookup.consultar(achado.documento)

            return replace(
                resultado,
                beneficiario=Beneficiario(
                    documento=achado.documento,
                    tipo=achado.tipo,
                    empresa=empresa,
                    confirmado_pelo_codigo=achado.confirmado_pelo_codigo,
                ),
            )
        except Exception:
            return resultado

    @staticmethod
    def _raiz_cnpj(boleto: Boleto) -> str | None:
        """Arrecadação segmento 6: posições 16-23 trazem a raiz do CNPJ."""
        if boleto.tipo != BoletoType.ARRECADACAO or not boleto.codigo_barras:
            return None
        if boleto.codigo_barras[1] != "6":
            return None
        return boleto.codigo_barras[15:23]

    # ---------------- Boleto ----------------
    def _processar(self, arquivo: Path, pagina: int | None, codigo: str) -> ExtractionResult:
        try:
            boleto = self._montar_boleto(codigo)
        except BoletoValidationError as exc:
            return ExtractionResult(
                arquivo=arquivo,
                status=ExtractionStatus.INVALID,
                mensagens=(str(exc),),
                pagina=pagina,
            )

        return ExtractionResult(
            arquivo=arquivo,
            status=ExtractionStatus.SUCCESS,
            boleto=boleto,
            confianca=1.0,
            mensagens=boleto.detalhes.observacoes if boleto.detalhes else (),
            pagina=pagina,
        )

    def _montar_boleto(self, codigo: str) -> Boleto:
        if codigo.startswith("8"):
            return Boleto(
                tipo=BoletoType.ARRECADACAO,
                codigo_barras=codigo,
                linha_digitavel=self._arrecadacao.converter(codigo),
                detalhes=self._parser.arrecadacao(codigo),
            )

import cv2
import numpy as np
import zxingcpp

from boleto_extractor.application.ports.barcode_reader import (
    BarcodeReading,
)
from boleto_extractor.core.exceptions import BarcodeReadingError
from boleto_extractor.core.types import ImageArray


class ZXingBarcodeReader:
    """Implementação do leitor de códigos de barras utilizando ZXing-C++."""

    def __init__(
        self,
        try_rotate: bool = True,
        try_downscale: bool = True,
        try_invert: bool = True,
    ) -> None:
        self._formats = zxingcpp.barcode_formats_from_str("ITF, Code128")
        self._try_rotate = try_rotate
        self._try_downscale = try_downscale
        self._try_invert = try_invert

    def ler(self, imagem: ImageArray) -> tuple[BarcodeReading, ...]:
        self._validar_imagem(imagem)

        try:
            imagem_bgr = cv2.cvtColor(imagem, cv2.COLOR_RGB2BGR)
            imagem_bgr = np.ascontiguousarray(imagem_bgr)

            codigos = zxingcpp.read_barcodes(
                imagem_bgr,
                formats=self._formats,
                try_rotate=self._try_rotate,
                try_downscale=self._try_downscale,
                try_invert=self._try_invert,
            )

            return tuple(
                BarcodeReading(
                    valor=codigo.text,
                    formato=str(codigo.format),
                )
                for codigo in codigos
                if codigo.valid and codigo.text
            )

        except Exception as exc:
            raise BarcodeReadingError(
                "Não foi possível realizar a leitura do código de barras."
            ) from exc

    def _validar_imagem(self, imagem: ImageArray) -> None:
        if not isinstance(imagem, np.ndarray):
            raise BarcodeReadingError(
                "A imagem precisa ser um array NumPy."
            )

        if imagem.size == 0:
            raise BarcodeReadingError("A imagem está vazia.")

        if imagem.dtype != np.uint8:
            raise BarcodeReadingError(
                "A imagem precisa utilizar o tipo uint8."
            )

        if imagem.ndim != 3 or imagem.shape[2] != 3:
            raise BarcodeReadingError(
                "A imagem precisa possuir três canais RGB."
            )

from __future__ import annotations

import re
import threading
from dataclasses import dataclass
from typing import Any, Callable, Iterable

TABELA_PADRAO = "opencnpj-bigquery.public.receita"
_PADRAO_CNPJ = re.compile(r"^[A-Z0-9]{12}\d{2}$")


def normalizar_cnpj(valor: str | None) -> str:
    """Remove máscara e aceita CNPJ numérico ou alfanumérico (Receita, jul/2026)."""
    limpo = re.sub(r"[^A-Z0-9]", "", (valor or "").upper())
    return limpo if _PADRAO_CNPJ.match(limpo) else ""


class LimiteExcedido(RuntimeError):
    def __init__(self, gb: float, limite: int) -> None:
        super().__init__(
            f"Consulta estimada em {gb:.2f} GB, acima do limite de {limite} GB. "
            f"Aumente o limite para pelo menos {int(gb) + 1} GB."
        )
        self.gb = gb


@dataclass(frozen=True)
class DadosCnpjBigQuery:
    cnpj: str
    razao_social: str
    nome_fantasia: str
    situacao: str
    municipio: str = ""
    uf: str = ""
    cnae_principal: str = ""
    fonte: str = "BigQuery"


class BigQueryCnpjClient:
    """Consulta CNPJ na base pública OpenCNPJ via BigQuery."""

    def __init__(self, obter_config: Callable[[], dict[str, Any]]) -> None:
        self._obter_config = obter_config
        self._cliente = None
        self._assinatura: tuple | None = None
        self._cache: dict[str, DadosCnpjBigQuery | None] = {}
        self._lock = threading.Lock()
        self.ultimo_gb: float = 0.0

    # ---------------- Público ----------------
    def ativo(self) -> bool:
        cfg = self._obter_config()
        return bool(cfg.get("bq_ativo") and cfg.get("bq_credencial"))

    def consultar(self, cnpj: str) -> DadosCnpjBigQuery | None:
        cnpj = normalizar_cnpj(cnpj)
        if not cnpj:
            return None
        return self.consultar_lote([cnpj]).get(cnpj)

    def consultar_lote(self, cnpjs: Iterable[str]) -> dict[str, DadosCnpjBigQuery | None]:
        """Uma única consulta para vários CNPJs."""
        limpos = {c for c in (normalizar_cnpj(x) for x in cnpjs) if c}
        if not limpos or not self.ativo():
            return {}

        with self._lock:
            faltando = sorted(c for c in limpos if c not in self._cache)
            if faltando:
                encontrados = self._consultar_bq(faltando)
                for c in faltando:
                    self._cache[c] = encontrados.get(c)
            return {c: self._cache[c] for c in limpos}

    # ---------------- Interno ----------------
    @staticmethod
    def _tabela(cfg: dict[str, Any]) -> str:
        tabela = (cfg.get("bq_tabela") or TABELA_PADRAO).strip().strip("`")
        if not re.fullmatch(r"[\w\-]+\.[\w\-]+\.[\w\-]+", tabela):
            raise ValueError(f"Nome de tabela inválido: {tabela}")
        return tabela

    @staticmethod
    def _limite_gb(cfg: dict[str, Any]) -> int:
        return int(cfg.get("bq_limite_gb") or 5)

    def _cliente_bq(self, cfg: dict[str, Any]):
        from google.cloud import bigquery
        from google.oauth2 import service_account

        assinatura = (cfg["bq_credencial"], cfg.get("bq_projeto"), self._tabela(cfg))
        if self._cliente is None or assinatura != self._assinatura:
            cred = service_account.Credentials.from_service_account_file(cfg["bq_credencial"])
            self._cliente = bigquery.Client(
                credentials=cred, project=cfg.get("bq_projeto") or cred.project_id
            )
            self._assinatura = assinatura
            self._cache.clear()
        return self._cliente

    def _estimar_gb(self, cliente, sql: str, params: list) -> float:
        """Dry-run: calcula os bytes sem executar e sem cobrar."""
        from google.cloud import bigquery

        job = cliente.query(
            sql,
            job_config=bigquery.QueryJobConfig(
                query_parameters=params, dry_run=True, use_query_cache=False
            ),
        )
        return (job.total_bytes_processed or 0) / 1024**3

    def _consultar_bq(self, cnpjs: list[str]) -> dict[str, DadosCnpjBigQuery]:
        from google.cloud import bigquery

        cfg = self._obter_config()
        cliente = self._cliente_bq(cfg)

        sql = f"""
            SELECT cnpj, razao_social, nome_fantasia, situacao_cadastral,
                   uf, municipio, cnae_principal
            FROM `{self._tabela(cfg)}`
            WHERE cnpj IN UNNEST(@cnpjs)
        """
        params = [bigquery.ArrayQueryParameter("cnpjs", "STRING", cnpjs)]

        limite = self._limite_gb(cfg)
        self.ultimo_gb = self._estimar_gb(cliente, sql, params)
        if self.ultimo_gb > limite:
            raise LimiteExcedido(self.ultimo_gb, limite)

        job_config = bigquery.QueryJobConfig(
            query_parameters=params,
            maximum_bytes_billed=max(1, limite) * 1024**3,
            use_query_cache=True,
        )
        resultado: dict[str, DadosCnpjBigQuery] = {}
        for r in cliente.query(sql, job_config=job_config).result():
            cnpj = normalizar_cnpj(r.cnpj)
            if not cnpj or cnpj in resultado:
                continue
            resultado[cnpj] = DadosCnpjBigQuery(
                cnpj=cnpj,
                razao_social=(r.razao_social or "").strip(),
                nome_fantasia=(r.nome_fantasia or "").strip(),
                situacao=(r.situacao_cadastral or "").strip().upper(),
                municipio=(r.municipio or "").strip(),
                uf=(r.uf or "").strip().upper(),
                cnae_principal=(r.cnae_principal or "").strip(),
            )
        return resultado

"""Lookup de CNPJ composto: tenta o OpenCNPJ primeiro e usa o BigQuery como reserva.
O cliente BigQuery pode ser trocado em tempo de execução (via configurações)."""
from __future__ import annotations

import logging
import threading

from boleto_extractor.application.ports.cnpj_lookup import CnpjLookup
from boleto_extractor.domain.entities.beneficiario import DadosEmpresa

log = logging.getLogger(__name__)


class CompositeCnpjLookup:
    """Implementa o Protocol CnpjLookup. Nunca lança exceção: retorna None em falha."""

    def __init__(self, primario: CnpjLookup, bigquery: CnpjLookup | None = None) -> None:
        self._primario = primario
        self._bigquery = bigquery
        self._lock = threading.Lock()  # set_bigquery roda na thread da UI; consultar, em workers
        self._cache: dict[str, DadosEmpresa] = {}

    def set_bigquery(self, cliente: CnpjLookup | None) -> None:
        with self._lock:
            self._bigquery = cliente
        log.info("BigQuery %s.", "ativado" if cliente else "desativado")

    def consultar(self, cnpj: str) -> DadosEmpresa | None:
        cnpj = "".join(c for c in cnpj if c.isdigit())
        if not cnpj:
            return None

        with self._lock:
            if cnpj in self._cache:
                return self._cache[cnpj]
            bigquery = self._bigquery

        for nome, fonte in (("OpenCNPJ", self._primario), ("BigQuery", bigquery)):
            if fonte is None:
                continue
            if nome == "BigQuery" and not getattr(fonte, "disponivel", True):
                continue
            try:
                dados = fonte.consultar(cnpj)
            except Exception as exc:  # noqa: BLE001 - contrato: nunca lançar
                log.warning("%s falhou para %s: %s", nome, cnpj, exc)
                dados = None
            if dados is not None:
                with self._lock:
                    self._cache[cnpj] = dados
                return dados

        return None

from __future__ import annotations

import logging
from typing import Any, Protocol

log = logging.getLogger(__name__)


class CnpjLookup(Protocol):
    def consultar(self, cnpj: str) -> Any | None: ...


class FallbackCnpjLookup:
    """Tenta cada fonte em ordem; a primeira que responder vence."""

    def __init__(self, *fontes: CnpjLookup) -> None:
        self._fontes = fontes

    def consultar(self, cnpj: str) -> Any | None:
        for fonte in self._fontes:
            if hasattr(fonte, "ativo") and not fonte.ativo():
                continue
            try:
                resultado = fonte.consultar(cnpj)
                if resultado:
                    return resultado
            except Exception as exc:  # uma fonte falhar não derruba a extração
                log.warning("Fonte %s falhou para %s: %s", type(fonte).__name__, cnpj, exc)
        return None

    def __getattr__(self, nome: str):
        # Repassa outros métodos (ex.: fechar sessão) para a fonte principal
        return getattr(self._fontes[0], nome)

import json
import threading
import time
import urllib.error
import urllib.request
from typing import Any

from boleto_extractor.application.services.documento_validator import cnpj_valido, limpar
from boleto_extractor.core.config import settings
from boleto_extractor.domain.entities.beneficiario import DadosEmpresa
from boleto_extractor.infrastructure.cnpj.sqlite_cnpj_cache import (
    STATUS_NAO_ENCONTRADO,
    STATUS_OK,
    SqliteCnpjCache,
)


class OpenCnpjClient:
    """Consulta a OpenCNPJ com cache em memória + SQLite e controle de limite (429)."""

    def __init__(self, cache: SqliteCnpjCache | None = None) -> None:
        self._cache = cache or SqliteCnpjCache()
        self._memoria: dict[str, DadosEmpresa | None] = {}
        self._lock = threading.Lock()
        self._ultima = 0.0
        self._bloqueado_ate = 0.0

    def consultar(self, cnpj: str) -> DadosEmpresa | None:
        try:
            c = limpar(cnpj)
            if not cnpj_valido(c):
                return None

            with self._lock:  # serializa: o mesmo CNPJ nunca é consultado em paralelo
                if c in self._memoria:
                    return self._memoria[c]

                em_cache = self._cache.obter(c)
                if em_cache is not None:
                    self._memoria[c] = em_cache[1]
                    return em_cache[1]

                resposta = self._requisitar(c)
                if resposta is None:
                    return None  # falha transitória: não cacheia, tenta depois

                status, dados = resposta
                self._cache.salvar(c, status, dados)
                self._memoria[c] = dados
                return dados
        except Exception:
            return None

    def _requisitar(self, cnpj: str) -> tuple[str, DadosEmpresa | None] | None:
        if time.time() < self._bloqueado_ate:
            return None

        espera = settings.opencnpj_intervalo_min - (time.time() - self._ultima)
        if espera > 0:
            time.sleep(espera)

        req = urllib.request.Request(
            settings.opencnpj_url.format(cnpj),
            headers={"Accept": "application/json", "User-Agent": "ExtratorBoletos/1.0"},
        )
        try:
            with urllib.request.urlopen(req, timeout=settings.opencnpj_timeout) as r:
                return STATUS_OK, self._converter(json.loads(r.read().decode("utf-8")))
        except urllib.error.HTTPError as e:
            if e.code == 404:
                return STATUS_NAO_ENCONTRADO, None
            if e.code == 429:
                self._bloqueado_ate = time.time() + settings.opencnpj_pausa_429
            return None
        except Exception:
            return None
        finally:
            self._ultima = time.time()

    @staticmethod
    def _converter(j: dict[str, Any]) -> DadosEmpresa:
        def s(chave: str) -> str:
            return str(j.get(chave) or "")

        telefone = ""
        for t in j.get("telefones") or []:
            if isinstance(t, dict) and t.get("numero") and not t.get("is_fax"):
                ddd = str(t.get("ddd") or "")
                telefone = f"({ddd}) {t['numero']}" if ddd else str(t["numero"])
                break

        return DadosEmpresa(
            razao_social=s("razao_social"),
            nome_fantasia=s("nome_fantasia"),
            situacao_cadastral=s("situacao_cadastral"),
            municipio=s("municipio"),
            uf=s("uf"),
            email=s("email").lower(),
            telefone=telefone,
            cnae_principal=s("cnae_principal"),
            porte_empresa=s("porte_empresa"),
            opcao_simples=s("opcao_simples"),
            opcao_mei=s("opcao_mei"),
        )

            arq = caminho or settings.data_dir / "cache_cnpj.sqlite"
            arq.parent.mkdir(parents=True, exist_ok=True)
            self._con = sqlite3.connect(arq, check_same_thread=False)
            self._con.execute(
                "CREATE TABLE IF NOT EXISTS cnpj ("
                " cnpj TEXT PRIMARY KEY, status TEXT NOT NULL,"
                " dados TEXT, consultado_em REAL NOT NULL)"
            )
            self._con.commit()
        except Exception:
            self._con = None

    def obter(self, cnpj: str) -> tuple[str, DadosEmpresa | None] | None:
        """None = não está em cache (ou expirou)."""
        if self._con is None:
            return None
        try:
            with self._lock:
                row = self._con.execute(
                    "SELECT status, dados, consultado_em FROM cnpj WHERE cnpj = ?", (cnpj,)
                ).fetchone()
            if not row:
                return None
            status, dados, quando = str(row[0]), row[1], float(row[2])
            ttl = settings.cache_ttl_ok if status == STATUS_OK else settings.cache_ttl_nao_encontrado
            if time.time() - quando > ttl:
                return None
            if status != STATUS_OK:
                return status, None
            bruto = json.loads(dados or "{}")
            return status, DadosEmpresa(**{k: str(v) for k, v in bruto.items() if k in _CAMPOS})
        except Exception:
            return None

    def salvar(self, cnpj: str, status: str, dados: DadosEmpresa | None) -> None:
        if self._con is None:
            return
        try:
            payload = json.dumps(asdict(dados) if dados else {}, ensure_ascii=False)
            with self._lock:
                self._con.execute(
                    "INSERT OR REPLACE INTO cnpj VALUES (?, ?, ?, ?)",
                    (cnpj, status, payload, time.time()),
                )
                self._con.commit()
        except Exception:
            pass

_DOIS = Decimal("0.01")

_SALDO = (
    "(p.pagparc_vlrdocumento + COALESCE(p.pagparc_vlracrescimo,0)"
    " - COALESCE(p.pagparc_vlrdesconto,0) - COALESCE(p.pagparc_vlrbaixas,0)"
    " - COALESCE(p.pagparc_vlrpago,0))"

@dataclass(frozen=True, slots=True)
class ParcelaErp:
    cod: int
    conta: int
    parcela: int
    qtd_parcelas: int
    vencimento: date | None
    valor_documento: Decimal
    saldo: Decimal
    pago: bool
    nome: str
    nota: str

    @property
    def rotulo(self) -> str:
        return f"Parc. {self.parcela}/{self.qtd_parcelas or '?'} (#{self.cod})"


class ErpParcelaRepository:
    """Somente leitura no ERP."""

    def __init__(self, config: Callable[[], dict[str, Any]], senha: Callable[[], str]) -> None:
        self._config = config
        self._senha = senha

    def _conectar(self):
        import pymysql

        cfg = self._config()
        return pymysql.connect(
            host=cfg["db_host"], port=int(cfg["db_porta"]), user=cfg["db_usuario"],
            password=self._senha(), database=cfg["db_banco"] or None,
            charset="latin1", ssl={"ssl": {}} if cfg.get("db_ssl") else None,
            connect_timeout=5, read_timeout=30,
        )

    def parcelas_das_contas(self, contas: Iterable[int]) -> dict[int, list[ParcelaErp]]:
        contas = sorted({int(c) for c in contas})
        resultado: dict[int, list[ParcelaErp]] = defaultdict(list)
        if not contas:
            return {}

        marcas = ",".join(["%s"] * len(contas))
        sql = f"""
            SELECT p.pagparc_cod, p.pagparc_conta, p.pagparc_parcela, pg.pag_qtdparcelas,
                   p.pagparc_dtvencimento, p.pagparc_vlrdocumento, {_SALDO},
                   p.pagparc_dtpagamento, COALESCE(p.pagparc_nome, ''),
                   COALESCE(NULLIF(p.pagparc_nota, ''), pg.pag_numnf, '')
              FROM pagar_parcelas p
              JOIN pagar pg ON pg.pag_cod = p.pagparc_conta
             WHERE p.pagparc_conta IN ({marcas})
          ORDER BY p.pagparc_conta, p.pagparc_dtvencimento, p.pagparc_parcela
        """
        with self._conectar() as con, con.cursor() as cur:
            cur.execute(sql, contas)
            for r in cur.fetchall():
                saldo = Decimal(str(r[6] or 0)).quantize(_DOIS)
                resultado[r[1]].append(ParcelaErp(
                    cod=r[0], conta=r[1], parcela=r[2] or 0, qtd_parcelas=r[3] or 0,
                    vencimento=r[4] if isinstance(r[4], date) else None,
                    valor_documento=Decimal(str(r[5] or 0)).quantize(_DOIS),
                    saldo=saldo,
                    pago=r[7] is not None or saldo <= Decimal("0.009"),
                    nome=(r[8] or "").strip(), nota=(r[9] or "").strip(),
                ))
        return dict(resultado)

from typing import cast

import cv2
import numpy as np

from boleto_extractor.core.exceptions import DocumentLoadError
from boleto_extractor.core.types import ImageArray


class ImageProcessor:
    """Realiza operações de preparação de imagens para reconhecimento."""

    def validar(self, imagem: ImageArray) -> None:
        if not isinstance(imagem, np.ndarray):
            raise DocumentLoadError(
                "A imagem precisa ser um array NumPy."
            )

        if imagem.size == 0:
            raise DocumentLoadError(
                "A imagem está vazia."
            )

        if imagem.dtype != np.uint8:
            raise DocumentLoadError(
                "A imagem precisa utilizar o tipo uint8."
            )

        if imagem.ndim != 3 or imagem.shape[2] != 3:
            raise DocumentLoadError(
                "A imagem precisa possuir três canais RGB."
            )

    def redimensionar(
        self,
        imagem: ImageArray,
        largura_maxima: int = 2000,
    ) -> ImageArray:

        self.validar(imagem)

        altura, largura = imagem.shape[:2]

        if largura <= largura_maxima:
            return imagem

        proporcao = largura_maxima / largura

        nova_altura = int(altura * proporcao)

        return cast(
    ImageArray,
    cv2.resize(
        imagem,
        (largura_maxima, nova_altura),
        interpolation=cv2.INTER_AREA,
    ),
)
    def escala_cinza(
        self,
        imagem: ImageArray,
    ) -> ImageArray:

        self.validar(imagem)

        return cast(
    ImageArray,
    cv2.cvtColor(imagem, cv2.COLOR_RGB2GRAY),
)
    def binarizar(
        self,
        imagem: ImageArray,
    ) -> ImageArray:

        cinza = self.escala_cinza(imagem)

        return cast(
            ImageArray,
    cv2.adaptiveThreshold(
        cinza,
        255,
        cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
        cv2.THRESH_BINARY,
        31,
        11,
    ),
)

    def melhorar_contraste(
        self,
        imagem: ImageArray,
    ) -> ImageArray:

        cinza = self.escala_cinza(imagem)

        clahe = cv2.createCLAHE(
            clipLimit=2.0,
            tileGridSize=(8, 8),
        )

        return cast(
            ImageArray,
            clahe.apply(cinza),
        )

from collections.abc import Iterator
from pathlib import Path

import cv2
import numpy as np
import pymupdf

from boleto_extractor.core.config import settings
from boleto_extractor.core.exceptions import DocumentLoadError
from boleto_extractor.core.types import ImageArray
from boleto_extractor.infrastructure.pdf.pdf_senha import senhas_pdf


class LocalDocumentLoader:
    """Carrega documentos PDF e imagens diretamente na memória."""

    def __init__(self, dpi: int | None = None) -> None:
        self._dpi = dpi or settings.dpi

    def carregar(self, arquivo: Path) -> Iterator[ImageArray]:
        arquivo = arquivo.resolve()

        self._validar_arquivo(arquivo)

        extensao = arquivo.suffix.lower()

        if extensao == ".pdf":
            yield from self._carregar_pdf(arquivo)
        else:
            yield self._carregar_imagem(arquivo)

    def _validar_arquivo(self, arquivo: Path) -> None:
        if not arquivo.exists():
            raise DocumentLoadError(f"Arquivo não encontrado: {arquivo}")

        if not arquivo.is_file():
            raise DocumentLoadError(f"O caminho informado não é um arquivo: {arquivo}")

        if arquivo.suffix.lower() not in settings.supported_extensions:
            raise DocumentLoadError(f"Formato não suportado: {arquivo.suffix}")

    def _carregar_pdf(self, arquivo: Path) -> Iterator[ImageArray]:
        try:
            with senhas_pdf.abrir(arquivo) as documento:
                if documento.page_count == 0:
                    raise DocumentLoadError("O documento PDF não possui páginas.")

                for pagina in documento:
                    pixmap = pagina.get_pixmap(
                        dpi=self._dpi,
                        colorspace=pymupdf.csRGB,
                        alpha=False,
                    )
                    imagem = np.frombuffer(pixmap.samples, dtype=np.uint8)
                    imagem = imagem.reshape(pixmap.height, pixmap.width, 3)
                    yield np.ascontiguousarray(imagem)

        except DocumentLoadError:
            raise  # inclui PdfProtegidoError

        except Exception as exc:
            raise DocumentLoadError(
                f"Não foi possível processar o PDF: {arquivo.name}"
            ) from exc

    def _carregar_imagem(self, arquivo: Path) -> ImageArray:
        try:
            dados = np.fromfile(arquivo, dtype=np.uint8)
            imagem = cv2.imdecode(dados, cv2.IMREAD_COLOR)

            if imagem is None:
                raise DocumentLoadError(f"Não foi possível decodificar: {arquivo.name}")

            imagem_rgb = cv2.cvtColor(imagem, cv2.COLOR_BGR2RGB)
            return np.ascontiguousarray(imagem_rgb)

        except DocumentLoadError:
            raise

        except Exception as exc:
            raise DocumentLoadError(f"Falha ao carregar a imagem: {arquivo.name}") from exc

from __future__ import annotations

import json
import threading
from pathlib import Path

import pymupdf

from boleto_extractor.core.exceptions import DocumentLoadError

_SERVICO = "BoletoExtractor-PDF"
_CHAVE = "senhas"
_MAX_LEMBRADAS = 30


class PdfProtegidoError(DocumentLoadError):
    """Nenhuma senha informada pelo usuário abre o PDF."""

    def __init__(self, arquivo: Path) -> None:
        super().__init__(f"PDF protegido por senha: {arquivo.name}")
        self.arquivo = arquivo


class SenhasPdf:
    """
    Guarda apenas senhas digitadas pelo usuário:
    - por arquivo (memória, só nesta sessão);
    - "usar nos próximos" (memória, só nesta sessão);
    - "lembrar" (cofre do Windows, somente se o usuário marcar).
    Nunca grava senha em log, CSV ou configuração exportada.
    """

    def __init__(self) -> None:
        self._lock = threading.Lock()
        self._por_arquivo: dict[str, str] = {}
        self._sessao: list[str] = []
        self._lembradas: list[str] = self._ler_cofre()

    @staticmethod
    def _chave(arquivo: Path | str) -> str:
        return str(Path(arquivo).resolve()).lower()

    # ---------------- Cofre do Windows ----------------
    @staticmethod
    def _ler_cofre() -> list[str]:
        try:
            import keyring
            return list(json.loads(keyring.get_password(_SERVICO, _CHAVE) or "[]"))
        except Exception:
            return []

    def _gravar_cofre(self) -> None:
        try:
            import keyring
            keyring.set_password(_SERVICO, _CHAVE, json.dumps(self._lembradas[:_MAX_LEMBRADAS]))
        except Exception:
            pass

    def esquecer_lembradas(self) -> None:
        with self._lock:
            self._lembradas = []
            self._sessao = []
            self._gravar_cofre()

    # ---------------- Abertura ----------------
    def _candidatas(self, arquivo: Path | str) -> list[str]:
        with self._lock:
            todas = [self._por_arquivo.get(self._chave(arquivo)), *self._sessao, *self._lembradas]
        return [s for s in dict.fromkeys(todas) if s]

    def abrir(self, arquivo: Path | str) -> pymupdf.Document:
        """Abre o PDF; se protegido, usa só senhas informadas. Lança PdfProtegidoError."""
        doc = pymupdf.open(arquivo)  # type: ignore[no-untyped-call]
        if not doc.needs_pass:
            return doc
        for senha in self._candidatas(arquivo):
            if doc.authenticate(senha):
                with self._lock:
                    self._por_arquivo[self._chave(arquivo)] = senha
                return doc
        doc.close()
        raise PdfProtegidoError(Path(arquivo))

    def precisa_senha(self, arquivo: Path | str) -> bool:
        if Path(arquivo).suffix.lower() != ".pdf":
            return False
        try:
            self.abrir(arquivo).close()
            return False
        except PdfProtegidoError:
            return True
        except Exception:
            return False  # outros erros são reportados pelo loader

    def informar(
        self,
        arquivo: Path | str,
        senha: str,
        usar_nos_proximos: bool = False,
        lembrar: bool = False,
    ) -> bool:
        """Testa a senha digitada. Só guarda se ela abrir o arquivo."""
        if not senha:
            return False
        try:
            with pymupdf.open(arquivo) as doc:  # type: ignore[no-untyped-call]
                if doc.needs_pass and not doc.authenticate(senha):
                    return False
        except Exception:
            return False

        with self._lock:
            self._por_arquivo[self._chave(arquivo)] = senha
            if usar_nos_proximos and senha not in self._sessao:
                self._sessao.insert(0, senha)
            if lembrar:
                self._lembradas = [senha, *(s for s in self._lembradas if s != senha)]
                self._gravar_cofre()
        return True


senhas_pdf = SenhasPdf()

from __future__ import annotations

from dataclasses import dataclass

from PySide6.QtCore import QAbstractListModel, QByteArray, QModelIndex, Qt

from boleto_extractor.application.services.matching_service import ROTULOS, ResultadoCruzamento


@dataclass(slots=True)
class LinhaBoleto:
    arquivo: str
    codigo_barras: str
    linha_digitavel: str
    valor: Decimal | None
    vencimento: date | None = None
    origem_vencimento: str = ""     # 'codigo' | 'pdf' | 'manual' | ''
    documento: int | None = None
    resultado: ResultadoCruzamento | None = None
    erro_documento: bool = False
    erro_data: bool = False

    @property
    def tipo(self) -> str:
        return "Arrecadação" if self.codigo_barras.startswith("8") else "Cobrança"


def _moeda(v: Decimal | None) -> str:
    if v is None:
        return "--"
    return f"R$ {v:,.2f}".replace(",", "X").replace(".", ",").replace("X", ".")


class BoletoListModel(QAbstractListModel):
    _NOMES = [
        "arquivo", "tipo", "linhaDigitavel", "valor", "vencimento", "origemVencimento",
        "documento", "status", "statusTexto", "parcela", "detalhe", "erroDocumento", "erroData",
    ]
    ROLES = {Qt.UserRole + i + 1: n for i, n in enumerate(_NOMES)}

    def __init__(self, parent=None) -> None:
        super().__init__(parent)
        self._itens: list[LinhaBoleto] = []

    # ---------- Qt ----------
    def rowCount(self, parent=QModelIndex()) -> int:
        return 0 if parent.isValid() else len(self._itens)

    def roleNames(self) -> dict[int, QByteArray]:
        return {r: QByteArray(n.encode()) for r, n in self.ROLES.items()}

    def data(self, index: QModelIndex, role: int = Qt.DisplayRole):
        if not index.isValid():
            return None
        it = self._itens[index.row()]
        nome = self.ROLES.get(role)
        res = it.resultado
        match nome:
            case "arquivo": return it.arquivo
            case "tipo": return it.tipo
            case "linhaDigitavel": return it.linha_digitavel
            case "valor": return _moeda(it.valor)
            case "vencimento": return it.vencimento.strftime("%d/%m/%Y") if it.vencimento else ""
            case "origemVencimento": return it.origem_vencimento
            case "documento": return str(it.documento) if it.documento else ""
            case "status": return res.status.value if res else ""
            case "statusTexto": return ROTULOS[res.status] if res else "Pendente"
            case "parcela": return res.parcela.rotulo if res and res.parcela else ""
            case "detalhe": return res.detalhe if res else ""
            case "erroDocumento": return it.erro_documento
            case "erroData": return it.erro_data
        return None

    # ---------- API ----------
    @property
    def itens(self) -> list[LinhaBoleto]:
        return self._itens

    def adicionar(self, linhas: list[LinhaBoleto]) -> None:
        existentes = {i.codigo_barras for i in self._itens}
        novas = [l for l in linhas if l.codigo_barras not in existentes]
        if not novas:
            return
        ini = len(self._itens)
        self.beginInsertRows(QModelIndex(), ini, ini + len(novas) - 1)
        self._itens.extend(novas)
        self.endInsertRows()

    def remover(self, row: int) -> None:
        if 0 <= row < len(self._itens):
            self.beginRemoveRows(QModelIndex(), row, row)
            del self._itens[row]
            self.endRemoveRows()

    def limpar(self) -> None:
        self.beginResetModel()
        self._itens.clear()
        self.endResetModel()

    def atualizar(self, row: int) -> None:
        idx = self.index(row)
        self.dataChanged.emit(idx, idx)

    def atualizar_todos(self) -> None:
        if self._itens:
            self.dataChanged.emit(self.index(0), self.index(len(self._itens) - 1))

from datetime import date
from decimal import Decimal


def formatar_linha(linha: str) -> str:
    """Formata a linha digitável no padrão visual dos boletos."""
    if len(linha) == 47:
        return (
            f"{linha[0:5]}.{linha[5:10]} {linha[10:15]}.{linha[15:21]} "
            f"{linha[21:26]}.{linha[26:32]} {linha[32]} {linha[33:]}"
        )
    if len(linha) == 48:
        return " ".join(f"{linha[i:i + 11]}-{linha[i + 11]}" for i in range(0, 48, 12))
    return linha


def formatar_valor(valor: Decimal | None) -> str:
    if valor is None:
        return ""
    texto = f"{valor:,.2f}".replace(",", "X").replace(".", ",").replace("X", ".")
    return f"R$ {texto}"


def formatar_valor_csv(valor: Decimal | None) -> str:
    return "" if valor is None else f"{valor:.2f}".replace(".", ",")


def formatar_data(data: date | None, estimada: bool = False) -> str:
    if data is None:
        return ""
    texto = data.strftime("%d/%m/%Y")
    return f"~{texto}" if estimada else texto

    function testarConsultaCnpj() {
        const cnpj = cnpjTeste.text.toUpperCase().replace(/[^A-Z0-9]/g, "")
        if (!/^[A-Z0-9]{12}\d{2}$/.test(cnpj)) {
            root.okBq = false
            root.msgBq = "Informe um CNPJ válido (14 caracteres, numérico ou alfanumérico)."
            return
        }
        root.bqTestando = true
        root.okBq = true
        root.msgBq = "Consultando…"
        settings.testarConsultaCnpj(cnpj)
    }

from typing import Any

from PySide6.QtCore import Property, QObject, QSettings, Qt, Signal, Slot
from PySide6.QtGui import QGuiApplication

from boleto_extractor.presentation.themes.catalog import SYSTEM_ID, THEMES, resolve_theme

_CHAVE = "aparencia/tema"


class ThemeManager(QObject):
    """Expõe o tema ativo ao QML e reage a mudanças do sistema operacional."""

    themeChanged = Signal()

    def __init__(self, app: QGuiApplication, parent: QObject | None = None) -> None:
        super().__init__(parent)
        self._settings = QSettings()
        self._hints = app.styleHints()

        selecionado = str(self._settings.value(_CHAVE, SYSTEM_ID))
        self._selecionado = selecionado if selecionado in THEMES else SYSTEM_ID

        self._hints.colorSchemeChanged.connect(self._ao_mudar_sistema)
        self._dados: dict[str, Any] = {}
        self._aplicar()

    def _sistema_escuro(self) -> bool:
        return bool(self._hints.colorScheme() == Qt.ColorScheme.Dark)

    def _aplicar(self) -> None:
        self._dados = resolve_theme(self._selecionado, self._sistema_escuro()).to_qml()
        self.themeChanged.emit()

    def _ao_mudar_sistema(self, *_: object) -> None:
        if self._selecionado == SYSTEM_ID:
            self._aplicar()

    @Property("QVariantMap", notify=themeChanged)
    def t(self) -> dict[str, Any]:
        return self._dados

    @Property(str, notify=themeChanged)
    def selecionado(self) -> str:
        return self._selecionado

    @Property("QVariantList", notify=themeChanged)
    def opcoes(self) -> list[dict[str, str]]:
        modo = "escuro" if self._sistema_escuro() else "claro"
        return [{"id": SYSTEM_ID, "nome": f"Seguir o sistema ({modo})"}] + [
            {"id": t.id, "nome": t.nome} for t in THEMES.values()
        ]

    @Slot(str)
    def selecionar(self, theme_id: str) -> None:
        if theme_id != SYSTEM_ID and theme_id not in THEMES:
            return
        self._selecionado = theme_id
        self._settings.setValue(_CHAVE, theme_id)
        self._aplicar()

class WorkerSignals(QObject):
    concluido = Signal(int, object)  # (job_id, tuple[ExtractionResult, ...])
    protegido = Signal(int, str)     # (job_id, caminho)


class ExtractionWorker(QRunnable):
    """Executa a extração fora da thread da interface."""

    def __init__(self, job_id: int, arquivo: Path, use_case: ExtractBoletoUseCase) -> None:
        super().__init__()
        self.setAutoDelete(False)
        self.signals = WorkerSignals()
        self._job_id = job_id
        self._arquivo = arquivo
        self._use_case = use_case

    def run(self) -> None:
        # PDF protegido sem senha conhecida: vai para a fila de senha, sem extrair
        if senhas_pdf.precisa_senha(self._arquivo):
            self.signals.protegido.emit(self._job_id, str(self._arquivo))
            return

        try:
            resultados = self._use_case.executar(self._arquivo)
        except Exception as exc:  # noqa: BLE001 - a interface nunca pode quebrar
            resultados = (
                ExtractionResult(
                    arquivo=self._arquivo,
                    status=ExtractionStatus.ERROR,
                    mensagens=(f"Erro inesperado: {exc}",),
                ),
            )
        self.signals.concluido.emit(self._job_id, resultados)

# ---------------- Utilitários de data/valor ----------------
def _parse_data(texto: str) -> date | None:
    dig = re.sub(r"\D", "", texto or "")
    if len(dig) != 8:
        return None
    try:
        return datetime.strptime(dig, "%d%m%Y").date()
    except ValueError:
        return None


def _para_date(valor: object) -> date | None:
    if isinstance(valor, datetime):
        return valor.date()
    if isinstance(valor, date):
        return valor
    return _parse_data(str(valor)) if valor else None


def _para_decimal(valor: object) -> Decimal | None:
    if valor is None or valor == "":
        return None
    try:
        return Decimal(str(valor)).quantize(Decimal("0.01"))
    except Exception:  # noqa: BLE001
        return None


def vencimento_efetivo(item: LinhaResultado) -> date | None:
    """Manual tem prioridade; senão, o vencimento extraído do código/PDF."""
    return item.vencimento_manual or _para_date(item.vencimento)

        self.notificacao.emit(f"✔ Boleto adicionado {valor}".strip())
        return True

    # ---------------- Entrada por arquivos ----------------
    @Slot("QVariantList")
    def adicionarArquivos(self, urls: list[object]) -> None:
        arquivos = self._expandir(urls)
        if not arquivos:
            self.notificacao.emit("Nenhum arquivo suportado (PDF, PNG, JPG).")
            return

        for arquivo in arquivos:
            job_id = self._novo_id()
            self._model.adicionar(job_id, arquivo)
            self._iniciar(job_id, arquivo)

        self.estadoChanged.emit()

    def _iniciar(self, job_id: int, arquivo: Path) -> None:
        worker = ExtractionWorker(job_id, arquivo, self._use_case)
        worker.signals.concluido.connect(self._ao_concluir)
        worker.signals.protegido.connect(self._ao_protegido)
        self._workers[job_id] = worker
        self._total += 1
        self._pool.start(worker)

    def _expandir(self, urls: list[object]) -> list[Path]:
        extensoes = set(settings.supported_extensions)
        arquivos: list[Path] = []

        for item in urls:
            url = item if isinstance(item, QUrl) else QUrl(str(item))
            caminho = Path(url.toLocalFile()) if url.isLocalFile() else Path(str(item))

            if caminho.is_dir():
                arquivos.extend(
                    sorted(p for p in caminho.rglob("*") if p.suffix.lower() in extensoes)
                )
            elif caminho.suffix.lower() in extensoes:
                arquivos.append(caminho)

        return arquivos

    def _contar_concluido(self) -> None:
        self._concluidos += 1
        if self._concluidos >= self._total:
            self._total = self._concluidos = 0
            self.notificacao.emit("Processamento concluído.")
        self.estadoChanged.emit()

    @Slot(int, object)
    def _ao_concluir(self, job_id: int, resultados: tuple[ExtractionResult, ...]) -> None:
        self._workers.pop(job_id, None)
        self._model.atualizar(job_id, resultados)
        self._contar_concluido()

    @Slot(int, str)
    def _ao_protegido(self, job_id: int, caminho: str) -> None:
        self._workers.pop(job_id, None)
        self._model.marcar_protegido(job_id)
        if all(j != job_id for j, _ in self._fila_senha):
            self._fila_senha.append((job_id, Path(caminho)))
            self.senhaChanged.emit()
        self._contar_concluido()

    # ---------------- Fila de senha ----------------
    def _reprocessar(self, job_id: int, arquivo: Path) -> None:
        self._model.reabrir(job_id)
        self._iniciar(job_id, arquivo)
        self.estadoChanged.emit()

    def _avancar(self) -> None:
        self._fila_senha.pop(0)
        self._atendidos += 1
        self._tentativas = 0
        self._erro_senha = ""
        if not self._fila_senha:
            self._atendidos = 0
        self.senhaChanged.emit()

    @Slot(str, bool, bool, result=bool)
    def informarSenha(self, senha: str, usar_nos_proximos: bool, lembrar: bool) -> bool:
        if not self._fila_senha:
            return False
        job_id, arquivo = self._fila_senha[0]

        if not senhas_pdf.informar(arquivo, senha, usar_nos_proximos, lembrar):
            self._tentativas += 1
            if self._tentativas >= _MAX_TENTATIVAS:
                self.notificacao.emit(f"🔒 {arquivo.name}: tentativas esgotadas. Arquivo pulado.")
                self._avancar()
            else:
                self._erro_senha = f"Senha incorreta ({self._tentativas}/{_MAX_TENTATIVAS})."
                self.senhaChanged.emit()
            return False

        self._avancar()
        self._reprocessar(job_id, arquivo)
        abertos = 1

        # A mesma senha pode abrir os próximos arquivos do mesmo emissor
        if usar_nos_proximos:
            for item in list(self._fila_senha):
                if not senhas_pdf.precisa_senha(item[1]):
                    self._fila_senha.remove(item)
                    self._atendidos += 1
                    self._reprocessar(*item)
                    abertos += 1
            if not self._fila_senha:
                self._atendidos = 0
            self.senhaChanged.emit()

        self.notificacao.emit(f"🔓 {abertos} arquivo(s) desbloqueado(s)")
        return True

    @Slot()
    def pularSenha(self) -> None:
        if self._fila_senha:
            self._avancar()

    @Slot()
    def pularTodasSenhas(self) -> None:
        self._fila_senha.clear()
        self._atendidos = 0
        self._tentativas = 0
        self._erro_senha = ""
        self.senhaChanged.emit()

    @Slot(int)
    def informarSenhaLinha(self, linha: int) -> None:
        """Chamado pelo botão 🔒 da tabela, para tentar de novo um arquivo pulado."""
        item = self._model.item(linha)
        if item is None or not item.protegido:
            return
        self._fila_senha = [(j, a) for j, a in self._fila_senha if j != item.job_id]
        self._fila_senha.insert(0, (item.job_id, item.caminho))
        self._tentativas = 0
        self._erro_senha = ""
        self.senhaChanged.emit()

    # ---------------- Edição para o cruzamento ----------------
    @Slot(int, str)
    def definirDocumento(self, linha: int, texto: str) -> None:
        item = self._model.item(linha)
        if item is None:
            return
        dig = re.sub(r"\D", "", texto or "")
        novo = int(dig) if dig else None
        campos: dict[str, object] = {"erro_documento": False}
        if novo != item.documento:
            campos |= {"documento": novo, "cruzamento": None}
        self._model.alterar(linha, **campos)

    @Slot(int, str)
    def definirVencimento(self, linha: int, texto: str) -> None:
        item = self._model.item(linha)
        if item is None or _para_date(item.vencimento):
            return  # vencimento vindo da extração fica travado
        digitos = re.sub(r"\D", "", texto or "")
        nova = _parse_data(texto)
        if digitos and nova is None:
            self._model.alterar(linha, erro_data=True)
            self.notificacao.emit(f"⚠ Data inválida na linha {linha + 1}")
            return
        campos: dict[str, object] = {"erro_data": nova is None}
        if nova != item.vencimento_manual:
            campos |= {"vencimento_manual": nova, "cruzamento": None}
        self._model.alterar(linha, **campos)

    @Slot(int, result=bool)
    def vencimentoEditavel(self, linha: int) -> bool:
        item = self._model.item(linha)
        return item is not None and _para_date(item.vencimento) is None

    # ---------------- Cruzamento ----------------
    @Slot()
    def cruzar(self) -> None:
        if self._repo is None:
            self.notificacao.emit("⚠ Configure a conexão com o ERP antes de cruzar.")
            return
        if self._cruzando or self.processando:
            return

        elegiveis = [
            (i, it) for i, it in enumerate(self._model.itens())
            if it.status == ExtractionStatus.SUCCESS and it.codigo
        ]
        if not elegiveis:
            self.notificacao.emit("Nenhum boleto válido para cruzar.")
            return

        pendentes = 0
        itens: list[ItemCruzamento] = []
        for linha, it in elegiveis:
            venc = vencimento_efetivo(it)
            valor = _para_decimal(it.valor)
            erro_doc, erro_data = it.documento is None, venc is None
            self._model.alterar(linha, erro_documento=erro_doc, erro_data=erro_data)
            if erro_doc or erro_data or valor is None:
                pendentes += 1
                continue
            itens.append(ItemCruzamento(it.codigo, it.documento, venc, valor))

        if pendentes:
            self.notificacao.emit(
                f"⚠ {pendentes} linha(s) sem documento ou data. Preencha os campos em vermelho."
            )
            return

        self._set_cruzando(True)
        self.notificacao.emit("Consultando ERP…")
        worker = _CruzamentoWorker(self._repo, itens)
        worker.signals.concluido.connect(self._ao_cruzar)
        self._worker_cruzamento = worker
        self._pool.start(worker)

    @Slot(object, str)
    def _ao_cruzar(self, resultados: list, erro: str) -> None:
        self._worker_cruzamento = None
        self._set_cruzando(False)
        if erro:
            self.notificacao.emit(f"⛔ {erro}")
            return
        self._model.aplicar_cruzamento({r.chave: r for r in resultados})
        ok = sum(1 for r in resultados if r.status == StatusCruzamento.ENCONTRADO)
        self.notificacao.emit(f"🔗 Cruzamento concluído: {ok}/{len(resultados)} encontrado(s)")
        self.estadoChanged.emit()

    # ---------------- Ações ----------------
    @Slot(str)
    def copiar(self, texto: str) -> None:
        if texto:
            QGuiApplication.clipboard().setText(texto)
            self.notificacao.emit("Linha digitável copiada!")

    @staticmethod
    def _linha_csv(item: LinhaResultado) -> list[object]:
        b = item.beneficiario
        e = item.empresa
        c = item.cruzamento
        venc = vencimento_efetivo(item)
        return [
            item.caminho.name, item.pagina or "", item.status, item.tipo,
            item.emissor, venc.strftime("%d/%m/%Y") if venc else "",
            formatar_valor_csv(item.valor),
            formatar_linha(item.linha), item.codigo, item.mensagem,
            b.documento_formatado if b else "", b.tipo if b else "",
            e.razao_social, e.nome_fantasia, e.situacao_cadastral, e.municipio, e.uf,
            e.email, e.telefone, e.cnae_principal, e.porte_empresa,
            e.opcao_simples, e.opcao_mei,
            item.documento or "",
            ROTULOS[c.status] if c else "",
            c.parcela.rotulo if c and c.parcela else "",
            c.detalhe if c else "",
        ]

    @Slot(QUrl)
    def exportarCsv(self, url: QUrl) -> None:
        destino = Path(url.toLocalFile())
        try:
            with destino.open("w", newline="", encoding="utf-8-sig") as f:
                escritor = csv.writer(f, delimiter=";")
                escritor.writerow(_CABECALHO)
                for item in self._model.itens():
                    escritor.writerow(self._linha_csv(item))
            self.notificacao.emit(f"Exportado para {destino.name}")
        except OSError as exc:
            self.notificacao.emit(f"Falha ao exportar: {exc}")

    @Slot()
    def limpar(self) -> None:
        if not self.processando and not self._cruzando:
            self._model.limpar()
            self.pularTodasSenhas()
            self.estadoChanged.emit()

@dataclass(slots=True)

    job_id: int
    caminho: Path
    pagina: int | None = None
    status: str = ExtractionStatus.PENDING.value
    tipo: str = ""
    emissor: str = ""
    vencimento: str = ""
    valor: Decimal | None = None

    @property

    def detalhe_beneficiario(self) -> str:

            return ""
        e = self.empresa
        local = " / ".join(x for x in (e.municipio, e.uf) if x)
        pares = (
            (b.tipo, b.documento_formatado),
            ("Razão social", e.razao_social),
            ("Fantasia", e.nome_fantasia),
            ("Situação", e.situacao_cadastral),

            ("CNAE", e.cnae_principal),
            ("Porte", e.porte_empresa),
            ("Simples", e.opcao_simples),
            ("MEI", e.opcao_mei),
        )
        linhas = [f"{k}: {v}" for k, v in pares if v]
        if b.confirmado_pelo_codigo:
            linhas.append("✔ CNPJ confirmado pelo código de barras")
        if b.tipo == "CNPJ" and b.empresa is None:
            linhas.append("Dados da empresa indisponíveis no momento")
        return "\n".join(linhas)


class ResultListModel(QAbstractListModel):
    def __init__(self, parent: QObject | None = None) -> None:
        super().__init__(parent)
        self._itens: list[LinhaResultado] = []

    def rowCount(self, parent: QModelIndex | QPersistentModelIndex = QModelIndex()) -> int:
        return 0 if parent.isValid() else len(self._itens)

    def roleNames(self) -> dict[int, QByteArray]:
        return {_BASE + i: QByteArray(nome.encode()) for i, nome in enumerate(_CAMPOS)}

    def data(self, index: QModelIndex | QPersistentModelIndex, role: int = 0) -> object:
        if not index.isValid():
            return None
        posicao = role - _BASE
        if not 0 <= posicao < len(_CAMPOS):
            return None

        item = self._itens[index.row()]
        campo = _CAMPOS[posicao]
        if campo == "beneficiarioDetalhe":
            return item.detalhe_beneficiario()

        valores: dict[str, object] = {
            "arquivo": item.caminho.name,
            "caminho": str(item.caminho),
            "pagina": item.pagina or 0,
            "status": item.status,
            "tipo": item.tipo,
            "emissor": item.emissor,
            "vencimento": item.vencimento,
            "valor": formatar_valor(item.valor),
            "linha": item.linha,
            "linhaFormatada": formatar_linha(item.linha),
            "codigo": item.codigo,
            "mensagem": item.mensagem,
            "beneficiario": item.beneficiario.nome if item.beneficiario else "",
            "beneficiarioDoc": item.documento,
            "situacao": item.empresa.situacao_cadastral,
            "protegido": item.protegido,
        }
        return valores[campo]

    def _linha_do_job(self, job_id: int, **filtro: object) -> int | None:
        return next(
            (
                i for i, item in enumerate(self._itens)
                if item.job_id == job_id
                and all(getattr(item, k) == v for k, v in filtro.items())
            ),
            None,
        )

    def _notificar(self, linha: int) -> None:
        indice = self.index(linha)
        self.dataChanged.emit(indice, indice)

    def adicionar(self, job_id: int, caminho: Path) -> None:
        linha = len(self._itens)
        self.beginInsertRows(QModelIndex(), linha, linha)
        self._itens.append(LinhaResultado(job_id=job_id, caminho=caminho))
        self.endInsertRows()

    def adicionar_resultado(self, job_id: int, resultado: ExtractionResult) -> None:
        """Insere um resultado já processado (ex.: código bipado no leitor)."""
        item = LinhaResultado(job_id=job_id, caminho=resultado.arquivo)
        self._preencher(item, resultado)
        linha = len(self._itens)
        self.beginInsertRows(QModelIndex(), linha, linha)
        self._itens.append(item)
        self.endInsertRows()

    def atualizar(self, job_id: int, resultados: tuple[ExtractionResult, ...]) -> None:
        linha = self._linha_do_job(job_id, status=ExtractionStatus.PENDING.value)
        if linha is None or not resultados:
            return

        primeiro, *resto = resultados
        self._preencher(self._itens[linha], primeiro)
        self._notificar(linha)

        if resto:
            self.beginInsertRows(QModelIndex(), linha + 1, linha + len(resto))
            for deslocamento, resultado in enumerate(resto, start=1):
                item = LinhaResultado(job_id=job_id, caminho=resultado.arquivo)
                self._preencher(item, resultado)
                self._itens.insert(linha + deslocamento, item)
            self.endInsertRows()

    # ---------------- PDF protegido ----------------
    def marcar_protegido(self, job_id: int) -> None:
        linha = self._linha_do_job(job_id, status=ExtractionStatus.PENDING.value)
        if linha is None:
            return
        item = self._itens[linha]
        item.status = ExtractionStatus.REVIEW.value
        item.mensagem = MSG_PROTEGIDO
        item.protegido = True
        self._notificar(linha)

    def reabrir(self, job_id: int) -> None:
        """Volta o item protegido para a fila de processamento."""
        linha = self._linha_do_job(job_id, protegido=True)
        if linha is None:
            return
        item = self._itens[linha]
        item.status = ExtractionStatus.PENDING.value
        item.mensagem = "Na fila…"
        item.protegido = False
        self._notificar(linha)

    def item(self, linha: int) -> LinhaResultado | None:
        return self._itens[linha] if 0 <= linha < len(self._itens) else None

    def _preencher(self, item: LinhaResultado, resultado: ExtractionResult) -> None:
        item.status = resultado.status.value
        item.pagina = resultado.pagina
        item.mensagem = " | ".join(resultado.mensagens)
        item.beneficiario = resultado.beneficiario
        item.protegido = False

        boleto = resultado.boleto
        if boleto is None:
            return

        item.tipo = TIPOS[boleto.tipo]
        item.linha = boleto.linha_digitavel or ""
        item.codigo = boleto.codigo_barras or ""

        if boleto.detalhes:
            d = boleto.detalhes
            item.emissor = d.emissor
            item.valor = d.valor
            item.vencimento = formatar_data(d.vencimento, d.vencimento_estimado)

    def contem_codigo(self, codigo: str) -> bool:
        return any(item.codigo == codigo for item in self._itens)

    def limpar(self) -> None:
        self.beginResetModel()
        self._itens.clear()
        self.endResetModel()

    def contar(self, status: ExtractionStatus) -> int:
        return sum(1 for item in self._itens if item.status == status.value)

    def total_valor(self) -> Decimal:
        return sum(
            (i.valor for i in self._itens
             if i.valor is not None and i.status == ExtractionStatus.SUCCESS.value),
            Decimal(0),
        )

    def itens(self) -> tuple[LinhaResultado, ...]:
        return tuple(self._itens)

def calcular_modulo_10(numero: str) -> int:
    soma = 0
    peso = 2

    for digito in reversed(numero):
        resultado = int(digito) * peso
        soma += resultado // 10 + resultado % 10
        peso = 1 if peso == 2 else 2

    return (10 - soma % 10) % 10

def _mod11(numero: str) -> int:
    soma, peso = 0, 2
    for d in reversed(numero):
        soma += int(d) * peso
        peso = 2 if peso == 9 else peso + 1
    r = 11 - soma % 11
    return 1 if r in (0, 10, 11) else r


def montar_cobranca(
    banco: str = "237", fator: str = "1000", valor: str = "0000015050", livre: str = "1" * 25
) -> str:
    sem_dv = banco + "9" + fator + valor + livre
    return sem_dv[:4] + str(_mod11(sem_dv)) + sem_dv[4:]

class FakeReader:
    """Cada chamada a ler() devolve os códigos da próxima página."""

        valores = self._paginas.pop(0) if self._paginas else ()

    resolve_theme,
)

HEX = re.compile(r"^#[0-9A-Fa-f]{6}$")


def _cores(obj: Any) -> list[str]:
    valores: list[str] = []
    for campo in fields(obj):
        valor = getattr(obj, campo.name)
        if is_dataclass(valor):
            valores.extend(_cores(valor))
        elif isinstance(valor, str) and valor.startswith("#"):
            valores.append(valor)
    return valores

"""
Gera um único arquivo com a estrutura e o conteúdo do projeto,
pronto para ser enviado a uma IA.

Uso:
    python tools/gerar_snapshot.py
    python tools/gerar_snapshot.py --saida snapshot.txt --max-kb 200
"""

from __future__ import annotations

import argparse
from pathlib import Path

RAIZ = Path(__file__).resolve().parent.parent

PASTAS_IGNORADAS = {
    ".venv", "venv", ".git", "__pycache__", ".pytest_cache",
    ".mypy_cache", ".ruff_cache", ".idea", ".vscode",
    "build", "dist", "htmlcov", ".tox", "node_modules",
}

EXTENSOES_INCLUIDAS = {
    ".py", ".toml", ".cfg", ".ini", ".md", ".txt",
    ".yaml", ".yml", ".json", ".spec",
}

ARQUIVOS_IGNORADOS = {"snapshot_projeto.txt", ".coverage"}


def deve_ignorar(caminho: Path) -> bool:
    partes = set(caminho.relative_to(RAIZ).parts)
    if partes & PASTAS_IGNORADAS:
        return True
    if any(p.endswith(".egg-info") for p in partes):
        return True
    return caminho.name in ARQUIVOS_IGNORADOS


def gerar_arvore(arquivos: list[Path]) -> str:
    linhas = [f"{RAIZ.name}/"]
    for arquivo in arquivos:
        relativo = arquivo.relative_to(RAIZ)
        nivel = len(relativo.parts) - 1
        linhas.append(f"{'    ' * nivel}{relativo.name}")
    return "\n".join(linhas)


def coletar_arquivos() -> list[Path]:
    return sorted(
        p for p in RAIZ.rglob("*")
        if p.is_file()
        and not deve_ignorar(p)
        and p.suffix.lower() in EXTENSOES_INCLUIDAS
    )


def ler_conteudo(arquivo: Path, max_bytes: int) -> str:
    if arquivo.stat().st_size > max_bytes:
        return f"[arquivo omitido: maior que {max_bytes // 1024} KB]"
    try:
        return arquivo.read_text(encoding="utf-8")
    except UnicodeDecodeError:
        return arquivo.read_text(encoding="latin-1")


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--saida", default="snapshot_projeto.txt")
    parser.add_argument("--max-kb", type=int, default=200)
    args = parser.parse_args()

    arquivos = coletar_arquivos()
    saida = RAIZ / args.saida
    max_bytes = args.max_kb * 1024

    with saida.open("w", encoding="utf-8") as f:
        f.write("# ESTRUTURA DO PROJETO\n\n")
        f.write(gerar_arvore(arquivos))
        f.write("\n\n# CONTEÚDO DOS ARQUIVOS\n")

        for arquivo in arquivos:
            relativo = arquivo.relative_to(RAIZ).as_posix()
            f.write(f"\n\n{'=' * 80}\n")
            f.write(f"ARQUIVO: {relativo}\n")
            f.write(f"{'=' * 80}\n")
            f.write(ler_conteudo(arquivo, max_bytes))

    tamanho_kb = saida.stat().st_size / 1024
    print(f"✔ {len(arquivos)} arquivos -> {saida.name} ({tamanho_kb:.1f} KB)")


if __name__ == "__main__":
    main()
